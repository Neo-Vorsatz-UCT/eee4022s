\section{Computer Vision}
The computer vision pipeline of choice was inspired by Drăgoi \textit{et al.} \cite{bib:teleoperation_intuitive_using_MediaPipeHands}. OpenCV was used to extract the live video feed of the laptop's camera, which was then fed into Google's MediaPipe Hand Landmark Detection \cite{bib:MediaPipe_Hand_landmarks_detecion}. MediaPipe Hand Landmark Detection provides monocular hand tracking and pose estimation, and is the successor to MediaPipe Hands, as MediaPipe Hands is now deprecated.

Other computer vision pipelines that were considered include Ultralytics' YOLO \cite{bib:Ultralytics_YOLO} and MMPose. The \textbf{YOLO models} are extremely customisable and easily trainable, making them excellent for niche computer vision tasks such as forensics \cite{bib:YOLO_in_forensics}, and overall hand detection, given a good data set \cite{bib:YOLO_in_complex_background}. There are no pretrained models for hand pose estimation, and the time constraints of this project did not allow for training a custom model. \textbf{MMPose} has pretrained models for hand pose estimation, but isn't as accurate as MediaPipe for this specific task \cite{bib:MMPose_versus_MediaPipe}.

\section{Kinematic Mapping}

The inverse kinematics uses the FABRIK algorithm. In particular, it implements an adaptation of FABRIK described by Moreno \textit{et al.} \cite{bib:ik_FABRIK_constraints}, which incorporates movement for more joints (not just ball joints), as well as additional motion constraints. The pseudocode for the algorithm can be seen in \autoref{alg:fabrik}, while the pseudocode for each type of joint can be seen in \autoref{alg:move_hinge} - \autoref{alg:move_prismatic}.

\subsection{Algorithm}
This implementation uses a linked list structure to describe the kinematics and positions of an arbitrary serial manipulator. Like with normal FABRIK \cite{bib:ik_original_FABRIK}, the algorithm iterates through this linked list structure, adjusting positions of joints under the constraints of its previous joints. This process is repeated for both a forward-reaching and backward-reaching stage. The adapation of FABRIK restricts where successive joints can be positioned based on the type of joint that came before, and then corrects the position based on the joint's range of motion, such as [-90\degree, 90\degree].

\begin{algorithm}
\caption{FABRIK Algorithm}
\label{alg:fabrik}
\footnotesize{
\begin{algorithmic}[1]
    \Require A linked list $L$ of joints where $L[0]$ is the world frame, and $L[1]$ is the base, $L[n-1]$ is the end-effector, and $L[n]$ is the target, an acceptable squared error $e$, and a maximum number of forward-backward iterations $n$
    \Ensure The end-effector is within $e$ of the target, or iterations reach maximum
    
    \Function{Fabrik}{$L,e,n$}
        \LineComment[0]{Define world and target}
        \State $W\gets L[0]$
        \State $T\gets L[1]$
        \While{$T$ is not immovable}
            \State $T\gets T.child$
        \EndWhile
        \Statex

        \LineComment[0]{Loop until error reduced or iterations maxed}
        \State $error\gets |T.\vec{pos}-T.parent.\vec{pos}|^2$
        \State $i\gets 0$
        \While{$i<n$ \textbf{and} $error>e$}
            \LineComment[1]{Forward reaching}
            \State $J\gets T$
            \State $N\gets T.parent$
            \While{$N\neq W$}
                \State $P\gets J$
                \State $J\gets N$
                \State $N\gets N.parent$
                \If {$P$ is a hinge}
                    \State $\text{MoveHinge}(J, P, 1)$
                \ElsIf {$P$ is a pivot}
                    \State $\text{MovePivot}(J, P, N, 1)$
                \ElsIf {$P$ is prismatic \textbf{or} $P$ is immovable}
                    \State $\text{MovePrismatic}(J, P, 1)$
                \EndIf
            \EndWhile
            \Statex

            \LineComment[1]{Backward reaching}
            \State $J\gets W$
            \State $N\gets W.child$
            \While{$N\neq T$}
                \State $P\gets J$
                \State $J\gets N$
                \State $N\gets N.child$
                \If {$P$ is a hinge}
                    \State $\text{MoveHinge}(J, P, -1)$
                \ElsIf {$P$ is a pivot}
                    \State $\text{MovePivot}(J, P, N, -1)$
                \ElsIf {$P$ is prismatic \textbf{or} $P$ is immovable}
                    \State $\text{MovePrismatic}(J, P, -1)$
                \EndIf
            \EndWhile
            \Statex

            \LineComment[1]{End iteration}
            \State $error\gets |T.\vec{pos}-T.parent.\vec{pos}|^2$
            \State $i\gets i+1$
        \EndWhile
    \EndFunction
\end{algorithmic}
}
\end{algorithm}

\begin{algorithm}
\caption{Move Joint under Hinge Constraints}
\label{alg:move_hinge}
\footnotesize{
\begin{algorithmic}[1]
    \Require A pointer $J$ to a joint, a pointer $P$ to the previous joint, and the direction $dir$ of FABRIK (end2base$=1$, base2end$=-1$)
    \Ensure Joint $J$ is positioned \& oriented under the hinge constraints of $P$

    \Function{MoveHinge}{$J,P,dir$}
        \LineComment[0]{Project distance $\vec{d}$ onto the plane perpendicular to $P.\vec{axis}_{aux}$}
        \State $\vec{d}\gets P.\vec{pos}-J.\vec{pos}$
        \State $\vec{l}\gets\vec{d}-(\vec{d}\cdot P.\vec{axis}_{aux})P.\vec{axis}_{aux}$
        \Statex

        \LineComment[0]{Update joint orientation}
        \State $mag\_l \gets |\vec{l}|$
        \If {$mag\_l=0$}
            \State $J.\vec{axis}_{link}\gets P.\vec{axis}_{link}$
        \Else
            \State $J.\vec{axis}_{link}\gets (\vec{l}/mag\_l)dir$
        \EndIf
        \State $J.\vec{axis}_{aux}\gets P.\vec{axis}_{aux}$
        \Statex

        \LineComment[0]{Check joint limits}
        \State $sign\gets\sgn((J.\vec{axis}_{link}\times P.\vec{axis}_{link})\cdot J.\vec{axis}_{aux})dir$
        \State $P.value\gets\cos^{-1}(J.\vec{axis}_{link}\cdot P.\vec{axis}_{link})sign$
        \If {$P$ is limited}
            \If {$P.value>P.high$}
                \State $P.value\gets P.high$
                \State $J.\vec{axis}_{link}\gets P.\vec{axis}_{link}\cos(P.value)$
                \State $J.\vec{axis}_{link}\gets J.\vec{axis}_{link}+(P.\vec{axis}_{link}\times P.\vec{axis}_{aux})\sin(P.value)dir$
            \ElsIf {$P.value<P.low$}
                \State $P.value\gets P.low$
                \State $J.\vec{axis}_{link}\gets P.\vec{axis}_{link}\cos(P.value)$
                \State $J.\vec{axis}_{link}\gets J.\vec{axis}_{link}+(P.\vec{axis}_{link}\times P.\vec{axis}_{aux})\sin(P.value)dir$
            \EndIf
        \EndIf
        \Statex

        \LineComment[0]{Update joint position}
        \State $J.\vec{pos}\gets P.\vec{pos}-J.\vec{axis}_{link}J.len_{link}dir$
    \EndFunction
\end{algorithmic}
}
\end{algorithm}

\begin{algorithm}
\caption{Move Joint under Pivot Constraints}
\label{alg:move_pivot}
\footnotesize{
\begin{algorithmic}[1]
    \Require A pointer $J$ to a joint, a pointer $P$ to the previous joint, a pointer $N$ to the next joint, and the direction $dir$ of FABRIK (end2base$=1$, base2end$=-1$)
    \Ensure Joint $J$ is positioned \& oriented under the pivot constraints of $P$

    \Function{MovePivot}{$J,P,N,dir$}
        \LineComment[0]{Update joint position}
        \State $J.\vec{axis}_{link}\gets P.\vec{axis}_{link}$
        \State $J.\vec{pos}\gets P.\vec{pos}-J.\vec{axis}_{link}J.len_{link}dir$
        \Statex

        \LineComment[0]{Determine auxiliary axis direction}
        \State $\vec{d}\gets J.\vec{pos}-N.\vec{pos}$
        \State $\vec{a}\gets J.\vec{axis}_{link}\times\vec{d}$
        \State $mag\_a\gets|\vec{a}|$
        \If{$mag\_a=0$}
            \State $\vec{a}\gets P.\vec{axis}_{aux}$
        \Else
            \State $\vec{a}\gets\vec{a}/mag\_a$
        \EndIf
        \Statex

        \LineComment[0]{Correct the direction of the auxiliary axis}
        \State $J.\vec{axis}_{aux}\gets\vec{a}\sgn(J.\vec{axis}_{aux}\cdot\vec{a})$
        \Statex

        \LineComment[0]{Check joint limits}
        \State $sign\gets\sgn((J.\vec{axis}_{aux}\times P.\vec{axis}_{aux})\cdot J.\vec{axis}_{link})dir$
        \State $P.value\gets\cos^{-1}(J.\vec{axis}_{aux}\cdot P.\vec{axis}_{aux})sign$
        \If {$P$ is limited}
            \If {$P.value>P.high$}
                \State $P.value\gets P.high$
                \State $J.\vec{axis}_{aux}\gets P.\vec{axis}_{aux}\cos(P.value)$
                \State $J.\vec{axis}_{aux}\gets J.\vec{axis}_{aux}-(P.\vec{axis}_{link}\times P.\vec{axis}_{aux})\sin(P.value)dir$
            \ElsIf {$P.value<P.low$}
                \State $P.value\gets P.low$
                \State $J.\vec{axis}_{aux}\gets P.\vec{axis}_{aux}\cos(P.value)$
                \State $J.\vec{axis}_{aux}\gets J.\vec{axis}_{aux}-(P.\vec{axis}_{link}\times P.\vec{axis}_{aux})\sin(P.value)dir$
            \EndIf
        \EndIf
    \EndFunction
\end{algorithmic}
}
\end{algorithm}

\begin{algorithm}
\caption{Move Joint under Prismatic Constraints}
\label{alg:move_prismatic}
\footnotesize{
\begin{algorithmic}[1]
    \Require A pointer $J$ to a joint, a pointer $P$ to the previous joint, and the direction $dir$ of FABRIK (end2base$=1$, base2end$=-1$)
    \Ensure Joint $J$ is positioned \& oriented under the prismatic constraints of $P$

    \Function{MovePrismatic}{$J,P,dir$}
        \LineComment[0]{Update joint orientation}
        \State $J.\vec{axis}_{link}\gets P.\vec{axis}_{link}$
        \State $J.\vec{axis}_{aux}\gets P.\vec{axis}_{aux}$
        \Statex

        \LineComment[0]{Calculate joint length}
        \State $\vec{d}\gets P.\vec{pos}-J.\vec{pos}$
        \State $P.value\gets(\vec{d}\cdot J.\vec{axis}_{link})dir-J.len_{link}$
        \Statex

        \LineComment[0]{Check joint limits}
        \If {$P$ is limited}
            \If {$P.value>P.high$}
                \State $P.value\gets P.high$
            \ElsIf {$P.value<P.low$}
                \State $P.value\gets P.low$
            \EndIf
        \EndIf
        \Statex
        
        \LineComment[0]{Update joint position}
        \State $J.\vec{pos}\gets P.\vec{pos}-J.\vec{axis}_{link}(P.value+J.len_{link})dir$
    \EndFunction
\end{algorithmic}
}
\end{algorithm}
